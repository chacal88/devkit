// Task 1.3 of test-map-hub-links (design D-4): which functions of the two hub modules each linked test file executes beyond what loading
// the modules runs. V8 coverage (NODE_V8_COVERAGE) of `node --test` per file, spawned processes included.
// Usage: node coverage.mjs <worktree root> <out dir> [--baseline-only]
import { spawn, spawnSync } from 'node:child_process';
import { mkdirSync, mkdtempSync, readdirSync, readFileSync, realpathSync, rmSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { pathToFileURL } from 'node:url';

const [root, out, flag] = process.argv.slice(2);
const HUBS = ['plugins/openspec/hooks/change-loop.mjs', 'plugins/openspec/hooks/deliver-check.mjs'];
const URL_OF = Object.fromEntries(HUBS.map((h) => [h, pathToFileURL(join(root, h)).href]));
const POOL = 6;

// D-4's tables: the tests whose count beyond load is 0 (every other linked test is above 0).
const ZERO = {
  'plugins/openspec/hooks/change-loop.mjs': ['gen-agents', 'conduct-check', 'deliver-check-cli', 'deliver-check', 'flow-check', 'jira-task', 'merge-setup', 'release-merge', 'release-note', 'gh-version', 'github-setup'],
  'plugins/openspec/hooks/deliver-check.mjs': ['devkit-kit', 'compozy-driver', 'merge-setup', 'gh-version', 'github-setup'],
};

mkdirSync(out, { recursive: true });
const tmpBase = realpathSync('/var/tmp');

// Functions with a count above 0 in a coverage dir, per hub: Map<hub, Map<key, name>>.
function executed(dir) {
  const found = new Map(HUBS.map((h) => [h, new Map()]));
  let files = [];
  try { files = readdirSync(dir).filter((f) => f.endsWith('.json')); } catch { return found; }
  for (const f of files) {
    let data;
    try { data = JSON.parse(readFileSync(join(dir, f), 'utf8')); } catch { continue; }
    for (const script of data.result || []) {
      const hub = HUBS.find((h) => URL_OF[h] === script.url);
      if (!hub) continue;
      for (const fn of script.functions) {
        const r = fn.ranges[0];
        if (r.count > 0) found.get(hub).set(`${fn.functionName}@${r.startOffset}`, fn.functionName || '(anonymous)');
      }
    }
  }
  return found;
}

// 1. the baseline: what loading both modules runs.
const baseDir = join(out, 'baseline');
rmSync(baseDir, { recursive: true, force: true });
const base = spawnSync(process.execPath, ['--input-type=module', '-e', `await import(${JSON.stringify(URL_OF[HUBS[0]])}); await import(${JSON.stringify(URL_OF[HUBS[1]])})`], { cwd: root, env: { ...process.env, NODE_V8_COVERAGE: baseDir }, encoding: 'utf8' });
const baseline = executed(baseDir);
const topLevel = (hub) => [...baseline.get(hub).keys()].filter((k) => k.startsWith('@0')).length;
console.log(`baseline exit ${base.status}`);
for (const h of HUBS) console.log(`baseline ${h}: ${baseline.get(h).size} functions (${topLevel(h)} anonymous at offset 0)`);
if (flag === '--baseline-only') process.exit(base.status || 0);

// 2. the linked tests (35 distinct files of the two base lists), each alone, at most POOL at once.
const map = JSON.parse(readFileSync(join(root, 'scripts/test-map.json'), 'utf8'));
const tests = [...new Set(HUBS.flatMap((h) => map.files[h]))].sort();
console.log(`${tests.length} test files`);
const results = [];

function run(file) {
  return new Promise((resolve) => {
    const name = file.split('/').pop().replace(/\.test\.mjs$/, '');
    const cov = join(out, 'cov', name);
    rmSync(cov, { recursive: true, force: true });
    const tmp = mkdtempSync(join(tmpBase, 'tmhl-cov-'));
    const child = spawn(process.execPath, ['--test', '--test-reporter=dot', file], { cwd: root, env: { ...process.env, NODE_V8_COVERAGE: cov, TMPDIR: tmp }, stdio: ['ignore', 'pipe', 'pipe'] });
    let tail = '';
    for (const stream of [child.stdout, child.stderr]) stream.on('data', (chunk) => { tail = (tail + chunk).slice(-1500); });
    child.on('close', (code) => {
      const got = executed(cov);
      const entry = { name, file, exit: code, hubs: {} };
      for (const h of HUBS) {
        const beyond = [...got.get(h)].filter(([k]) => !baseline.get(h).has(k));
        entry.hubs[h] = { count: beyond.length, first: beyond.slice(0, 5).map(([, n]) => n) };
      }
      if (code !== 0) entry.tail = tail;
      results.push(entry);
      console.log(`done ${name} exit ${code} change-loop ${entry.hubs[HUBS[0]].count} deliver-check ${entry.hubs[HUBS[1]].count}`);
      rmSync(cov, { recursive: true, force: true });
      rmSync(tmp, { recursive: true, force: true });
      resolve();
    });
  });
}

const queue = [...tests];
await Promise.all(Array.from({ length: POOL }, async () => { while (queue.length) await run(queue.shift()); }));
results.sort((a, b) => a.file.localeCompare(b.file));
writeFileSync(join(out, 'results.json'), JSON.stringify({ baseline: Object.fromEntries(HUBS.map((h) => [h, baseline.get(h).size])), results }, null, 1));

// 3. the check of the task: exit codes, and each count on the side D-4's tables give it.
let bad = 0;
for (const h of HUBS) {
  for (const file of map.files[h]) {
    const r = results.find((x) => x.file === file);
    const name = r.name;
    const zero = ZERO[h].includes(name);
    const side = r.hubs[h].count === 0;
    if (r.exit !== 0) { bad++; console.log(`FAIL exit ${r.exit}: ${name}`); }
    if (zero !== side) { bad++; console.log(`DIFF ${h} ${name}: count ${r.hubs[h].count}, D-4 says ${zero ? '0' : 'above 0'}`); }
  }
}
console.log(`check: ${bad} difference(s)`);
process.exit(bad ? 1 : 0);
