// The read tracer of test-map-hub-links D-2. Loaded with NODE_OPTIONS=--import=<this file's URL>.
// TRACE_ROOT: the repository root; TRACE_LOG: the file each opened path under the root is appended to, relative to the root.
import fs from 'node:fs';
import fsp from 'node:fs/promises';
import { syncBuiltinESMExports } from 'node:module';
import { fileURLToPath } from 'node:url';
import { relative, resolve, isAbsolute } from 'node:path';

const root = resolve(process.env.TRACE_ROOT ?? '');
const log = process.env.TRACE_LOG;
const append = fs.appendFileSync;
let busy = false;

const record = (target) => {
  if (busy || typeof target === 'number' || !log) return;
  busy = true;
  try {
    const text = target instanceof URL ? fileURLToPath(target) : Buffer.isBuffer(target) ? target.toString() : String(target);
    const abs = resolve(text);
    const rel = relative(root, abs);
    if (rel !== '' && !rel.startsWith('..') && !isAbsolute(rel) && abs !== resolve(log)) append(log, `${rel}\n`);
  } catch { /* a path the tracer cannot read is not a document */ } finally { busy = false; }
};

const wrap = (owner, name) => {
  const original = owner[name];
  owner[name] = function traced(target, ...rest) { record(target); return original.call(this, target, ...rest); };
};

for (const name of ['readFileSync', 'readFile', 'openSync', 'open', 'createReadStream']) wrap(fs, name);
for (const name of ['readFile', 'open']) wrap(fsp, name);
syncBuiltinESMExports();

== the runner (run-trace.mjs), verbatim ==
// node run-trace.mjs <repo root> <out dir> <test file>...: each test file alone under the read tracer, one after the other.
import { spawn } from 'node:child_process';
import { mkdirSync, mkdtempSync, realpathSync, rmSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';

const [root, out, ...files] = process.argv.slice(2);
mkdirSync(out, { recursive: true });
const tracer = new URL('./trace-reads.mjs', import.meta.url).href;
const one = (file) => new Promise((done) => {
  const tag = file.replaceAll('/', '__');
  const tmp = realpathSync(mkdtempSync('/var/tmp/tmhl-trace-'));
  const env = { ...process.env, TMPDIR: tmp, TRACE_ROOT: root, TRACE_LOG: join(out, `${tag}.reads`), NODE_OPTIONS: `--import=${tracer}` };
  delete env.NODE_TEST_CONTEXT;
  const started = Date.now();
  const child = spawn(process.execPath, ['--test', '--test-reporter=dot', file], { cwd: root, env, stdio: ['ignore', 'pipe', 'pipe'] });
  let text = '';
  child.stdout.on('data', (c) => { text += c; });
  child.stderr.on('data', (c) => { text += c; });
  child.on('close', (code) => {
    writeFileSync(join(out, `${tag}.out`), text);
    writeFileSync(join(out, `${tag}.exit`), `${code}\n`);
    rmSync(tmp, { recursive: true, force: true });
    process.stdout.write(`${code === 0 ? 'pass' : 'FAIL'} ${((Date.now() - started) / 1000).toFixed(1)}s ${file}\n`);
    done();
  });
});
for (const file of files) await one(file);

slow = set(m['slow'])
CODE = ('.mjs', '.js', '.cjs', '.py', '.sh')


def is_document(key):
    return not key.startswith('plugins/') and not key.endswith(CODE) and '/bin/' not in '/' + key


doc_keys = sorted(k for k in m['files'] if is_document(k))
read = {}  # test file -> set of distinct repository paths opened
code = {}
for f in fast:
    p = os.path.join(out, tag(f))
    read[f] = set(l.strip() for l in open(p + '.reads')) if os.path.exists(p + '.reads') else set()
    code[f] = open(p + '.exit').read().strip()


def opens(f, key):
    return key in read[f] or (key.endswith('/') and any(r.startswith(key) for r in read[f]))


def short(f):
    return os.path.basename(f).removesuffix('.test.mjs')


if section == 'reads':
    for f in fast:
        keys = [k for k in doc_keys if opens(f, k)]
        print(f'{f}: exit {code[f]}, {len(read[f])} distinct repository paths opened, {len(keys)} of the {len(doc_keys)} document keys:')
        for k in keys:
            print(f'  {k}')

for k in doc_keys:

    if not any(t in slow for t in base):
