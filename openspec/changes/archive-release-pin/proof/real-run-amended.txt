== 1. The build and run script, verbatim (bash; WT is the worktree, P the scratch repository)

set -u
WT=/home/kaue/repos/devkit/.claude/worktrees/archive-release-pin
P=/var/tmp/devkit-archive-release-pin/proof-repo-amended
export TMPDIR=/var/tmp/devkit-archive-release-pin
export GIT_AUTHOR_NAME=proof GIT_AUTHOR_EMAIL=proof@example.invalid GIT_COMMITTER_NAME=proof GIT_COMMITTER_EMAIL=proof@example.invalid
H=$WT/plugins/openspec/hooks
LOOP=plugins/openspec/hooks/change-loop-compozy.test.mjs

echo '## Build'
set -x
mkdir -p $P
git init -q --bare -b main $P/origin.git
git clone -q $P/origin.git $P/seed
cd $P/seed
git checkout -q -b main
mkdir -p docs openspec/changes/demo plugins/openspec/hooks
git -C $WT show c561553^:docs/openspec-flow.md > docs/openspec-flow.md
git -C $WT show c561553^:plugins/openspec/hooks/change-loop.test.mjs > $LOOP
printf 'schema: devkit-lite\n' > openspec/config.yaml
printf 'schema: devkit-lite\ncreated: 2026-11-23\n' > openspec/changes/demo/.openspec.yaml
printf '# Demo\n' > openspec/changes/demo/proposal.md
printf '# Roadmap\n\n- [ ] demo\n' > openspec/roadmap.md
printf '# Contributing\n\nOriginal line.\n' > CONTRIBUTING.md
git add -A
git commit -q -m 'chore: scratch main'
git push -q origin main
set +x

# one fresh clone per case; the archive commit, then the case edit
mkcase() { # $1 = case id
  git clone -q $P/origin.git $P/$1
  cd $P/$1
  git checkout -q -b chore/archive-demo
  mkdir -p openspec/changes/archive
  git mv openspec/changes/demo openspec/changes/archive/2026-11-23-demo
  cat > openspec/changes/archive/2026-11-23-demo/retro.md <<'EOF'
# Retro

## Proposed updates

### `CONTRIBUTING.md` (docs)
- Answer: approved
- Applied: yes
EOF
  printf '# Contributing\n\nOriginal line.\nApplied proposal line.\n' > CONTRIBUTING.md
  printf '# Roadmap\n\n- [x] demo\n' > openspec/roadmap.md
}
after_doc()  { git -C $WT show c561553:docs/openspec-flow.md > docs/openspec-flow.md; }
after_test() { git -C $WT show c561553:plugins/openspec/hooks/change-loop.test.mjs > $LOOP; }
run() { # $1 = case id, $2 = hooks dir
  git add -A
  git commit -q -m "archive demo: $1"
  echo "--- $1"
  echo "cwd: $P/$1"
  echo "command: env GITHUB_BASE_REF=main GITHUB_HEAD_REF=chore/archive-demo node $2/deliver-check.mjs --ci --json"
  out=$(env GITHUB_BASE_REF=main GITHUB_HEAD_REF=chore/archive-demo node $2/deliver-check.mjs --ci --json 2>$P/$1.stderr)
  code=$?
  echo "exit code: $code"
  echo "archive.unnamed: $(printf '%s' "$out" | node -e 'let s="";process.stdin.on("data",d=>s+=d).on("end",()=>{const r=JSON.parse(s);console.log(JSON.stringify(r.archive?.unnamed))})')"
  echo "stderr: $(cat $P/$1.stderr)"
}

echo; echo '## Runs'
mkcase A1; after_doc; run A1 $H
mkcase A2; after_doc; sed -i '1s/$/ (one sentence edited)/' docs/openspec-flow.md; run A2 $H
mkcase A3; after_test; git diff --stat -- $LOOP | cat; run A3 $H
mkcase A4; git rm -q docs/openspec-flow.md; run A4 $H
mkdir -p $P/consumer-hooks
cp $H/guard.mjs $H/deliver-check.mjs $H/merge-setup.mjs $P/consumer-hooks/
echo "consumer folder: $(ls $P/consumer-hooks | tr '\n' ' ')"
mkcase A5; after_doc; run A5 $P/consumer-hooks

echo; echo '## A6 (in the worktree)'
cd $WT
PAT='[0-9]+\\?\.[0-9]+\\?\.[0-9]+'
echo "command: env TMPDIR=$TMPDIR GITHUB_HEAD_REF=probe-branch GITHUB_BASE_REF=main node scripts/test-suite.mjs --test-name-pattern=\"^change-loop docs: the design records the answers\$\" $LOOP"
env GITHUB_HEAD_REF=probe-branch GITHUB_BASE_REF=main node scripts/test-suite.mjs --test-name-pattern="^change-loop docs: the design records the answers$" $LOOP 2>&1 | grep "^✔\|^✖\|^ℹ \(tests\|pass\|fail\)"
echo "test-suite exit code: ${PIPESTATUS[0]}"
now=$(git grep -c -E "$PAT" -- $LOOP | cut -d: -f2)
was=$(git grep -c -E "$PAT" 34e3500 -- $LOOP | cut -d: -f3)
echo "command: git grep -c -E '$PAT' -- $LOOP   -> $now"
echo "command: git grep -c -E '$PAT' 34e3500 -- $LOOP   -> $was"
echo "drop: $((was - now))"
