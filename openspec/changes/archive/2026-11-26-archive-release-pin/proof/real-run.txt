printf '# Demo\n' > openspec/changes/demo/proposal.md
printf '# Roadmap\n\n- [ ] demo\n' > openspec/roadmap.md
printf '# Contributing\n\nOriginal line.\n' > CONTRIBUTING.md
git add -A
git commit -q -m 'chore: scratch main'
git push -q origin main
set +x

# one fresh clone per case; the archive commit, then the case edit
mkcase() { # $1 = case id
  git clone -q $P/origin.git $P/$1
  cd $P/$1
  git checkout -q -b chore/archive-demo
  git mv openspec/changes/demo openspec/changes/archive/2026-11-23-demo 2>/dev/null || { mkdir -p openspec/changes/archive; git mv openspec/changes/demo openspec/changes/archive/2026-11-23-demo; }
  cat > openspec/changes/archive/2026-11-23-demo/retro.md <<'EOF'
# Retro

## Proposed updates

### `CONTRIBUTING.md` (docs)
- Answer: approved
- Applied: yes
EOF
  printf '# Contributing\n\nOriginal line.\nApplied proposal line.\n' > CONTRIBUTING.md
  printf '# Roadmap\n\n- [x] demo\n' > openspec/roadmap.md
}
after_doc()  { git -C $WT show c561553:docs/openspec-flow.md > docs/openspec-flow.md; }
after_test() { git -C $WT show c561553:plugins/openspec/hooks/change-loop.test.mjs > plugins/openspec/hooks/change-loop.test.mjs; }
run() { # $1 = case id, $2 = hooks dir
  git add -A
  git commit -q -m "archive demo: $1"
  echo "--- $1"
  echo "cwd: $P/$1"
  echo "command: env GITHUB_BASE_REF=main GITHUB_HEAD_REF=chore/archive-demo node $2/deliver-check.mjs --ci --json"
  out=$(env GITHUB_BASE_REF=main GITHUB_HEAD_REF=chore/archive-demo node $2/deliver-check.mjs --ci --json 2>$P/$1.stderr)
  code=$?
  echo "exit code: $code"
  echo "archive.unnamed: $(printf '%s' "$out" | node -e 'let s="";process.stdin.on("data",d=>s+=d).on("end",()=>{const r=JSON.parse(s);console.log(JSON.stringify(r.archive?.unnamed))})')"
  echo "stderr: $(cat $P/$1.stderr)"
}

echo; echo '## Runs'
mkcase R1; after_doc; run R1 $H
mkcase R2; after_doc; sed -i '1s/$/ (one sentence edited)/' docs/openspec-flow.md; git diff --stat | cat; run R2 $H
mkcase R3; after_test; run R3 $H
mkcase R4; after_test
# one assert line under the pin lines (the block that holds the pin asserts)
ln=$(grep -n "assert.doesNotMatch(flow, /v0" plugins/openspec/hooks/change-loop.test.mjs | tail -1 | cut -d: -f1)
sed -i "${ln}a\\  assert.ok(flow.length > 0, 'x');" plugins/openspec/hooks/change-loop.test.mjs
git diff -U0 plugins/openspec/hooks/change-loop.test.mjs | tail -n +5 | cat
run R4 $H
mkcase R5; git rm -q docs/openspec-flow.md; run R5 $H
mkcase R6; after_doc; after_test; run R6 $H
mkdir -p $P/consumer-hooks
cp $H/guard.mjs $H/deliver-check.mjs $H/merge-setup.mjs $P/consumer-hooks/
echo "consumer folder: $(ls $P/consumer-hooks | tr '\n' ' ')"
mkcase R7; after_doc; after_test; run R7 $P/consumer-hooks
